// 붙여넣고 실행
(() => {
  // 여기 JSON 문자열을 네 설정으로 교체하거나 그대로 변수에 넣어라
  const cfg = {
    "fmkFilter::filterMode": "blur",
    "fmkFilter::hideRepliesCountMethod": "downvotes",
    "fmkFilter::hideRepliesMode": "blur",
    "fmkFilter::keywords": {
      "BJ": true,
      "bj": true
    },
    "fmkFilter::mainBanner": false,
    "fmkFilter::userReasons": {
      "다진버섯": "2026. 9. 18."
    },
    "fmkFilter::users": {
      "다진버섯": true
    }
  };

  // helper: 안전한 키워드 -> uBlock 라인 변환 (포텐 전용 + 제목 기준)
  function keywordToFilter(kw) {
    const esc = kw.replace(/\\/g,'\\\\').replace(/"/g,'\\"');
    // 포텐(메인)에서만 동작하고 제목 링크만 검사
    return `fmkorea.com##body:has(a[href*="/best"]) tr:has(a.title:has-text(${kw}))`;
  }

  // helper: 사용자(닉) -> uBlock 라인 (제목 옆 작성자 영역에 따라 달라질 수 있으니 이후 검증 필요)
  // 여기선 안전하게 "게시물 행에 닉네임 텍스트가 있는 경우"로 처리
  function userToFilter(user) {
    return `fmkorea.com##tr:has(a:has-text(${user}))`;
  }

  // collect rules
  const rules = [];
  rules.push('! === generated by config -> uBlock filters ===');
  rules.push('! paste this file into a .txt and host as RAW for uBlock/AdGuard list');

  // keywords
  if (cfg["fmkFilter::keywords"]) {
    Object.keys(cfg["fmkFilter::keywords"]).forEach(kw => {
      if (!cfg["fmkFilter::keywords"][kw]) return;
      // wrap in literal form if contains non-ascii or parentheses: use plain token
      rules.push(keywordToFilter(JSON.stringify(kw)));
    });
  }

  // users
  if (cfg["fmkFilter::users"]) {
    Object.keys(cfg["fmkFilter::users"]).forEach(u => {
      if (!cfg["fmkFilter::users"][u]) return;
      rules.push(userToFilter(JSON.stringify(u)));
    });
  }

  // output as single block
  const out = rules.join('\\n');
  console.log(out);
  // copy to clipboard in modern browsers:
  if (navigator.clipboard) navigator.clipboard.writeText(out).catch(()=>{});
  // also present the text in a window for manual copy
  const w = window.open('', '_blank');
  w.document.write('<pre>' + out.replace(/</g,'&lt;') + '</pre>');
})();
